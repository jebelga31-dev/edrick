


CREATE TABLE IF NOT EXISTS officer (
    OfficerID INT NOT NULL PRIMARY KEY,
    Firstname VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    gender VARCHAR(50) NOT NULL
);

CREATE TABLE IF NOT EXISTS station (
    stationID INT NOT NULL PRIMARY KEY,
    stationname VARCHAR(50) NOT NULL,
    district VARCHAR(50) NOT NULL
);

CREATE TABLE IF NOT EXISTS assignment (
    assignmentID INT NOT NULL PRIMARY KEY,
    officerID VARCHAR(50) NOT NULL,
    stationID VARCHAR(50) NOT NULL,
    caseID VARCHAR(50) NOT NULL,
    assignedDate VARCHAR(50) NOT NULL 
);

CREATE TABLE IF NOT EXISTS  vehicle (
    vehicleID INT NOT NULL PRIMARY KEY,
    platenumber VARCHAR(50) NOT NULL,
    stationID VARCHAR(50) NOT NULL,
    status VARCHAR(50) NOT NULL
);

CREATE TABLE IF NOT EXISTS  equipment (
    equipmentID INT NOT NULL PRIMARY KEY,
    itemtype VARCHAR(50) NOT NULL,
    modelandserial VARCHAR(50) NOT NULL,
    stationID VARCHAR(50) NOT NULL,
    status VARCHAR(50) NOT NULL
);
